import useAxios from "../../../Hooks/useAxios";
import axios from "axios";
import { useDispatch } from "react-redux";

import CLOSE_MOB_SIGNUP_MODAL from "../../../Redux/Action/MobAction/closemobsignupmodal.action";

import Cookies from "universal-cookie";

const TabSignup = () => {

  const [cityData, setCityData] = useState([]);

  const country = async () => {
    try {
      const response = await axios({
        method: "GET",
        url: "/countries.php?authkey=w98dfx4t321sfef9872r54inp54er43rgj0987s",
      });
      setCountries(response.data);
    } catch (err) {
      console.log(err);
    }
  };
  // End GET Country Name API

  // Start GET State Name API

    try {
      const response = await axios({
        method: "GET",
        url:

          value,
      });
      setStateData(response.data);
    } catch (err) {
      console.log(err);
    }
  };
  // End GET State Name API

  // Start GET City Name API

    try {
      const response = await axios({
        method: "GET",
        url:
          "/cities-v2.php?authkey=w98dfx4t321sfef9872r54inp54er43rgj0987s&state_id=" +
          value,
      });
      setCityData(response.data);
    } catch (err) {
      console.log(err);
    }
  };

    country(), state(1), city(1);

  const CountryId = (e) => {
    const value = e.target.value;
    state(value);
  };
  // End Country onChange function

    const value = e.target.value;
    city(value);
  };

        method: "POST",
        url:

          phone,
      });
    } catch (err) {
      console.log(err);
    }

    e.preventDefault();

        method: "POST",
        url: "/signup-v2.php?authkey=w98dfx4t321sfef9872r54inp54er43rgj0987s",

              color: "#c3dbf7",

              fontSize: "14px",

              {/* Start name coding */}
              <Form.Group className="mb-2" controlId="formBasicName">
                <Form.Label
                  className="mb-0 mt-2"
                  style={{
                    color: "#0b2299",
                    fontWeight: "500",
                    fontSize: "12px",
                  }}
                >
                  Name

                  required="required"

                    height: "25px",

                />
              </Form.Group>
              {/* Start name coding */}

              {/* Start password coding */}
              <Form.Group className="mb-3" controlId="formBasicPassword">

                  className="mb-0 mt-2"
                  style={{
                    color: "#0b2299",
                    fontWeight: "500",
                    fontSize: "12px",
                  }}

                    height: "25px",
                    borderRadius: "10px",
                    fontSize: "12px",
                  }}
                />
              </Form.Group>
              {/* Start passsword coding */}

              {/* Start phone coding */}
              <Form.Group className="mb-3" controlId="formBasicPassword">
                <Form.Label
                  className="mb-0 mt-2"
                  style={{
                    color: "#0b2299",
                    fontWeight: "500",
                    fontSize: "12px",
                  }}
                >
                  Phone Number
                </Form.Label>
                <Form.Control
                  type="number"
                  required="required"
                  placeholder="Enter Number"
                  name="phone"
                  style={{
                    height: "25px",
                    borderRadius: "10px",
                    fontSize: "12px",
                  }}
                />
              </Form.Group>
              {/* End phone coding */}

              <Button

                style={{ backgroundColor: "#6aadfe" }}

                  setFristSlide("d-none"), setSecondSlide("d-block");
                }}
              >
                NEXT
              </Button>
            </div>

            <div className={secondSlide}>
              {/* Start Email coding */}

                <Form.Label
                  className="mb-0 mt-2"
                  style={{
                    color: "#0b2299",
                    fontWeight: "500",
                    fontSize: "12px",
                  }}
                >
                  Email
                </Form.Label>
                <Form.Control

                    fontSize: "12px",

                />
              </Form.Group>
              {/*End Email coding */}

              {/* Start Your Profile */}
              <Form.Group className="mb-2" controlId="formBasicEmail">
                <Form.Label

                <Form.Select

                  name="account-type"
                  required="required"
                  style={{
                    height: "25px",

                  }}
                >
                  <option value="Money Changer">Money Changer</option>

                  <option value="Gold Trader">Gold Trader</option>
                </Form.Select>
              </Form.Group>
              {/* End Your Profile */}

                  className="mb-0 mt-2"
                  style={{ color: "#0b2299", fontWeight: "500" }}
                >
                  Company
                </Form.Label>
                <Form.Control
                  type="text"
                  placeholder="ForexBlues"

                    height: "25px",

                    borderColor: "#6aadfe",

                NEXT

            </div>

                </Form.Label>

                    height: "25px",

                      <Form.Label
                        className="mb-0 mt-2"

                      >

                        City

                          height: "25px",
